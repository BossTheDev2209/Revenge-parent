implement:

```lua
-- MentalService.luau — หลอดใจ: drain + delta จาก action (design doc §3)
-- option ข: ยังไม่มี drain modifiers — รอ user ตัดสินอายุ (flag ใน docs/08 §4)
-- bar เข้าระบบตอน phase 2 (Config.Mental.introducedInPhase) — ก่อนนั้น no-op ทั้งไฟล์
local isStudio = script ~= nil
local Config = if isStudio
	then require(game:GetService("ReplicatedStorage").Shared.Config)
	else require("../../shared/Config")

local MentalService = {}

-- 1%/4.5 วิจริง แปลงเป็นต่อชั่วโมงเกม (7.5 วิจริง)
MentalService.drainPerHour = Config.Mental.drainPercent
	* (Config.Time.secondsPerGameHour / Config.Mental.drainEverySeconds)

-- ค่า mental ที่สูตรควรใช้: ก่อน phase 2 คืน 50 (โซนกลาง ×1.0 — ไม่มีโบนัส/โทษ)
function MentalService.effective(state): number
	if state.phase < Config.Mental.introducedInPhase then return 50 end
	return state.mental
end

function MentalService.applyDelta(state, delta: number)
	if state.phase < Config.Mental.introducedInPhase then return end
	state.mental = math.clamp(state.mental + delta, 0, 100)
	if state.mental <= 0 then
		state.flags.mentalZero = true -- EndingService อ่านทีหลัง (Bad End 2)
	end
end

-- reason = key ใน Config.Mental.deltas (clipEdit / viralClip / rentMissed / ...)
function MentalService.apply(state, reason: string)
	local d = Config.Mental.deltas[reason]
	assert(d, "ไม่รู้จัก mental reason: " .. tostring(reason))
	MentalService.applyDelta(state, d)
end

function MentalService.onHour(state)
	MentalService.applyDelta(state, -MentalService.drainPerHour)
end

return MentalService
```

```lua
-- FollowerService.luau — ตัดคลิป(รับคะแนน) + อัปคลิป(ได้ follower/เงิน)
-- ไม่เรียก push / ไม่ require service อื่น — Main compose (mentalPct ส่งเข้ามาเป็น arg)
local isStudio = script ~= nil
local Config = if isStudio
	then require(game:GetService("ReplicatedStorage").Shared.Config)
	else require("../../shared/Config")
local Formulas = if isStudio
	then require(game:GetService("ReplicatedStorage").Shared.Formulas)
	else require("../../shared/Formulas")

local FollowerService = {}

-- client จบ QTE ส่งคะแนนมา (คะแนนไว้ใจ client ได้ — singleplayer)
function FollowerService.finishEdit(state, action)
	local score = math.clamp(tonumber(action.score) or 0, 0, Config.MaxClipAccuracy)
	table.insert(state.clips, {
		tier = Formulas.vidqTier(score),
		score = score,
		uploaded = false,
	})
end

-- อัปคลิป: follower = base(เฟสปัจจุบัน) × VidQ × mental แล้วอัปเดตเงิน/เฟส
-- คืน (สำเร็จไหม, viral ไหม) — viral แล้ว Main ไปบวก mental ต่อ
function FollowerService.uploadClip(state, action, mentalPct: number, rng)
	local clip = state.clips[tonumber(action.clip) or 0]
	if not clip or clip.uploaded then return false, false end
	local phase = Formulas.phaseFor(state.follower)
	local gained, viral = Formulas.followerGain(phase, clip.tier, mentalPct, rng)
	state.follower += gained
	state.money += Formulas.moneyGain(gained, phase)
	state.phase = Formulas.phaseFor(state.follower)
	clip.uploaded = true
	clip.gained = gained
	clip.viral = viral
	return true, viral
end

return FollowerService
```

```lua
-- MoneyService.luau — หักค่าใช้จ่ายราย 7 วันเกม (design doc §4)
-- เงินเข้าจากคลิปอยู่ FollowerService.uploadClip — ไฟล์นี้ดูแลรายจ่ายอย่างเดียว
local isStudio = script ~= nil
local Config = if isStudio
	then require(game:GetService("ReplicatedStorage").Shared.Config)
	else require("../../shared/Config")

local MoneyService = {}

function MoneyService.weeklyTotal(phase: number): number
	local w = Config.WeeklyExpenses[phase]
	return w.rent + w.internet + w.food
end

-- เรียกจาก TimeService.onDay (wiring ใน Main) — คืน true ถ้าจ่ายไม่ไหว (Main ไปหัก mental ต่อ)
function MoneyService.onDay(state): boolean
	if state.day <= 1 or (state.day - 1) % 7 ~= 0 then return false end
	local total = MoneyService.weeklyTotal(state.phase)
	if state.money >= total then
		state.money -= total
		return false
	end
	state.money = 0
	state.flags.rentMissed = (state.flags.rentMissed or 0) + 1 -- Bad End 1 = ครบ 2 (EndingService ทีหลัง)
	return true
end

return MoneyService
```

```lua
local MentalService = require(SSS.Services.MentalService)
local FollowerService = require(SSS.Services.FollowerService)
local MoneyService = require(SSS.Services.MoneyService)

-- ทุกการกระทำของผู้เล่น — service ไม่เรียก push เอง Main push หลัง route สำเร็จ
ActionRouter.register("FinishEdit", function(s, a)
	FollowerService.finishEdit(s, a)
	MentalService.apply(s, "clipEdit")
end)
ActionRouter.register("UploadClip", function(s, a)
	local ok, viral = FollowerService.uploadClip(s, a, MentalService.effective(s))
	if ok and viral then MentalService.apply(s, "viralClip") end
end)

RS.Shared.Remotes.Action.OnServerEvent:Connect(function(player, action)
	if ActionRouter.route(state, action) then GameState.push(state) end
end)  -- (แทน connect เดิม)

TimeService.onHour(function(s)
	MentalService.onHour(s)
	GameState.push(s)
end)  -- (แทน onHour เดิม)
TimeService.onDay(function(s)
	if MoneyService.onDay(s) then MentalService.apply(s, "rentMissed") end
end)
```
