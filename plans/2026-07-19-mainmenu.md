```lua
-- GameState: หน้า Main Menu ใช้ — Continue = slot ที่เซฟล่าสุด
function GameState.latestSlot(summaries)
	local best, bestAt = nil, -1
	for slot, sum in summaries do
		if sum.savedAt > bestAt then best, bestAt = slot, sum.savedAt end
	end
	return best
end

-- โหลด save ทับ state เดิมโดยรักษา identity ของตาราง (service ถือ reference เดิมอยู่)
function GameState.replaceWith(state, newState)
	for k in state do state[k] = nil end
	for k, v in newState do state[k] = v end
	return state
end

-- ActionRouter.route เพิ่ม arg 3: h(state, action, player)
```

## Task 2: Wiring actions ใน Main.server

```lua
ActionRouter.register("RequestSlots", function(s, a, player)
	s.slotList = GameState.listSlots(player.UserId)
end)
ActionRouter.register("LoadSlot", function(s, a, player)
	local loaded = GameState.loadSlot(player.UserId, tonumber(a.slot) or 1)
	if loaded then GameState.replaceWith(s, loaded) end
end)
ActionRouter.register("SaveSlot", function(s, a, player)
	s.slot = tonumber(a.slot) or s.slot
	GameState.saveSlot(player.UserId, s)
end)
ActionRouter.register("NewGame", function(s, a)
	local fresh = GameState.newGame()
	fresh.slot = tonumber(a.slot) or 1
	GameState.replaceWith(s, fresh)
end)
-- OnServerEvent ส่ง player เข้า route
```
